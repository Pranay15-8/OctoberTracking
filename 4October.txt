FIND THE NODES THAT ARE AT DISTANCE OF K FROM THE TARGET NODE-LEETCODE
------------------------------------------------------------
class Solution {
    public List<Integer> distanceK(TreeNode root, TreeNode target, int k) {
        Map<TreeNode,TreeNode> tm=new HashMap<>();
        findParent(root,tm);
        Set<TreeNode> visited=new HashSet<>();
        Queue<TreeNode> q=new LinkedList<>();
        int c=0;
        q.offer(target);
        visited.add(target);
        while(!q.isEmpty()){
            if(c==k)
            break;

            c++;
            int n=q.size();
            for(int i=0;i<n;i++){
                TreeNode t=q.poll();
                if(t.left!=null&&!visited.contains(t.left)){
                    q.offer(t.left);
                    visited.add(t.left);
                }
                if(t.right!=null&&!visited.contains(t.right)){
                    q.offer(t.right);
                    visited.add(t.right);
                }
                if(tm.get(t)!=null&&!visited.contains(tm.get(t))){
                    q.offer(tm.get(t));
                    visited.add(tm.get(t));
                }

            }
        }
        ArrayList<Integer> ans=new ArrayList<>();
        while(!q.isEmpty()){
            ans.add(q.poll().val);
        }
        return ans;
    }
    public void findParent(TreeNode root,Map<TreeNode,TreeNode> tm){
        Queue<TreeNode> q=new LinkedList<>();
        q.offer(root);
        while(!q.isEmpty()){
            TreeNode temp=q.poll();
            if(temp.left!=null){
                tm.put(temp.left,temp);
                q.offer(temp.left);
            }
            if(temp.right!=null){
                tm.put(temp.right,temp);
                q.offer(temp.right);
            }
        }
    }
}

class Solution {
    public int minTime(Node root, int target) {
        // code here
        Map<Node,Node> tm=new HashMap<>();
        Node tar=findParent(root,target,tm);
        int res=findMin(tar,tm);
        return res;
    }
    public int findMin(Node root,Map<Node,Node> tm){
        int d=0;
        Set<Node> hs=new HashSet<>();
        Queue<Node> q=new LinkedList<>();
        q.offer(root);
        hs.add(root);
        while(!q.isEmpty()){
            int n=q.size();
            int flag=0;
            for(int i=0;i<n;i++){
                root=q.poll();
                if(root.left!=null&&!hs.contains(root.left)){
                    flag=1;
                    hs.add(root.left);
                    q.add(root.left);
                }
                if(root.right!=null&&!hs.contains(root.right)){
                    flag=1;
                    hs.add(root.right);
                    q.add(root.right);
                }
                if(tm.get(root)!=null&&!hs.contains(tm.get(root))){
                    flag=1;
                    hs.add(tm.get(root));
                    q.add(tm.get(root));
                    
                }
            }
            if(flag==1)
            d++;
        }
        return d;
    }
    public Node findParent(Node root,int target,Map<Node,Node> tm){
        Queue<Node> q=new LinkedList<>();
        q.offer(root);
        Node temp=null;
        while(!q.isEmpty()){
            root=q.poll();
            if(root.data==target)
            temp=root;
            if(root.left!=null){
                tm.put(root.left,root);
                q.add(root.left);
            }
            if(root.right!=null){
                tm.put(root.right,root);
                q.add(root.right);
            }
        }
        return temp;
    }
}
