// BOTTOM VIEW OF BINARY TREE
class tuple{
    Node node;
    int v;
    public tuple(Node node,int v){
        this.node=node;
        this.v=v;
    }
}
class Solution {
    public ArrayList<Integer> bottomView(Node root) {
        ArrayList<Integer> ans=new ArrayList<>();
        if(root==null){
            return ans;
        }
     TreeMap<Integer,Integer> tm=new TreeMap<>();
     Queue<tuple> q=new LinkedList<>();
     q.offer(new tuple(root,0));
     while(!q.isEmpty()){
         tuple t=q.poll();
         Node n=t.node;
         int x=t.v;
         if(tm.get(x)==null)
         tm.put(x,n.data);
         else
         tm.put(x,n.data);
         
         if(n.left!=null){
             q.offer(new tuple(n.left,x-1));
         }
         if(n.right!=null){
             q.offer(new tuple(n.right,x+1));
         }
     }
     for(Map.Entry<Integer,Integer> entry:tm.entrySet()){
         ans.add(entry.getValue());
     }
     return ans;
        
    }
}

class Solution {
    public int findBottomLeftValue(TreeNode root) {  
        Queue<TreeNode> q=new LinkedList<>();
        int l=0;
        q.offer(root);
        while(!q.isEmpty()){
            int n=q.size();
            l=q.peek().val;
            for(int i=0;i<n;i++){
            root=q.poll();
            if(root.left!=null)
            q.offer(root.left);
            if(root.right!=null)
            q.offer(root.right);
            }  
        }
        return l;
    }
}
