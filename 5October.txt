
class Solution {
    public int countNodes(TreeNode root) {
        if(root==null)
        return 0;

        int l=getheightl(root);
        int r=getheightr(root);

        if(l==r)
        return (1<<r)-1;

        return 1+countNodes(root.left)+countNodes(root.right);
    }
    public int getheightl(TreeNode l){
       int h=0;
       while(l!=null){
        h++;
         l=l.left;
       }
       return h;
    }
    public int getheightr(TreeNode r){
        int h=0;
        while(r!=null){
            h++;
            r=r.right;
        }
        return h;
    }
}

class Solution {
    public int scoreOfParentheses(String s) {
        int res=0,d=0;
        for(int i=0;i<s.length();i++){
            if(s.charAt(i)=='(')
            d++;
            else{
                d--;
                if(s.charAt(i-1)=='(')
                res+=1<<d;
            }
        }
        return res;
    }

}

 public static int[][] fourSum(int[] arr, int target) { // Don't change the number of parameters
     // Please write your return statement here
     if(arr.length<4)
     return new int[][]{};
     Arrays.sort(arr);
    List<List<Integer>> ans=new ArrayList<>();
    int n=arr.length;
    for(int i=0;i<n-3;i++){
        if(i>0&&arr[i]==arr[i-1])
        continue;
        for(int j=i+1;j<n-2;j++){
            if(j>i+1&&arr[j]==arr[j-1])
            continue;
            int k=j+1,l=n-1;
            while(k<l){
                long sum=arr[i];
                sum+=arr[j];
                sum+=arr[k];
                sum+=arr[l];
                if(sum==target){
                    ArrayList<Integer> a=new ArrayList<>();
                    a.add(arr[i]);
                    a.add(arr[j]);
                    a.add(arr[k]);
                    a.add(arr[l]);
                    ans.add(new ArrayList<>(a));
                    k++;
                    l--;
                    while(k<l&&arr[k]==arr[k-1])
                    k++;
                    while(k<l&&arr[l]==arr[l+1])
                    l--;
                }
                if(sum<target){
                    k++;
                }else{
                    l--;
                }
            }
        }
    }
    int arrr[][]=new int[ans.size()][4];
    for(int i=0;i<arrr.length;i++){
        for(int j=0;j<4;j++){
            arrr[i][j]=ans.get(i).get(j);
        }
    }
     return arrr;
  }
