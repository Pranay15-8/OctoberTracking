
class Solution {
    public boolean checkTree(TreeNode root) {
        if(root==null)
        return true;

        if(root.left==null&&root.right==null)
        return true;

         int c=0;
         if(root.left!=null)
         c+=root.left.val;
         if(root.right!=null)
         c+=root.right.val;

         return (c==root.val)&&checkTree(root.left)&&checkTree(root.right);
        
    }
}

class Solution {
    public int deepestLeavesSum(TreeNode root) {
        Queue<TreeNode> q=new LinkedList<>();
        int last=0;
        q.offer(root);
        while(!q.isEmpty()){
            last=0;
            int n=q.size();
            for(int i=0;i<n;i++){
                root=q.poll();
                last+=root.val;
                if(root.left!=null)
                q.offer(root.left);
                if(root.right!=null)
                q.offer(root.right);
            }
        }
        return last;
    }
}

class Solution {
    public int longestValidParentheses(String s) {
       int ans=0;
       Stack<Integer> st=new Stack<>();
       st.push(-1);
       for(int i=0;i<s.length();i++){
          if(s.charAt(i)=='('){
            st.push(i);
          }else{
             st.pop();
             if(st.isEmpty())
             st.push(i);
             else
             ans=Math.max(ans,i-st.peek());
          }
       }
       return ans;
    }
}
