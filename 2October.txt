            int v=q.peek().data;
            int n=q.size();
            for(int i=0;i<n;i++){
                root=q.poll();
                if(root.left!=null)
                q.offer(root.left);
                if(root.right!=null)
                q.offer(root.right);
            }
            ans.add(v);
        }
        return ans;

class Solution {
    public ArrayList<ArrayList<Integer>> paths(Node root) {
        // code here
        ArrayList<ArrayList<Integer>> ans=new ArrayList<>();
        ArrayList<Integer> in=new ArrayList<>();
        dfs(root,ans,in);
        return ans;
    }
    public void dfs(Node root,ArrayList<ArrayList<Integer>> ans,ArrayList<Integer> in){
        if(root.left==null&&root.right==null){
            in.add(root.data);
            ans.add(new ArrayList<>(in));
            in.remove(in.size()-1);
            return;
        }
        in.add(root.data);
        if(root.left!=null)
        dfs(root.left,ans,in);
        if(root.right!=null)
        dfs(root.right,ans,in);
        in.remove(in.size()-1);
    }
}

 public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        if(root==null||p==root||q==root){
            return root;
        }
        TreeNode left=lowestCommonAncestor(root.left,p,q);
        TreeNode right=lowestCommonAncestor(root.right,p,q);

        if(left!=null&&right!=null)
        return root;
        return (left!=null)?left:right;
    }

 class tuple{
    TreeNode root;
    int d;
    public tuple(TreeNode root,int d){
        this.root=root;
        this.d=d;
    }

class Solution {
    public int widthOfBinaryTree(TreeNode root) {
        if(root==null)
        return 0;
        int ans=0;
        Queue<tuple> q=new LinkedList<>();
        q.offer(new tuple(root,0));
        while(!q.isEmpty()){
            int s=q.size();
            int last=0,first=0;
            int min=q.peek().d;
            for(int i=0;i<s;i++){
                int j=q.peek().d-min;
                TreeNode t=q.peek().root;
                q.poll();
               
                if(i==0)first=j;
                if(i==s-1)last=j;
                if(t.left!=null)
                q.offer(new tuple(t.left,j*2+1));
                if(t.right!=null)
                q.offer(new tuple(t.right,j*2+2));
            }
            ans=Math.max(ans,last-first+1);
        }
        return ans;
    }
}
