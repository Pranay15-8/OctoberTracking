 public static int threeSumClosest(int[] nums, int target) { 
     if(nums.length==3)
     return nums[0]+nums[1]+nums[2];
     Arrays.sort(nums);
     int cs=nums[0]+nums[1]+nums[nums.length-1];
     for(int i=0;i<nums.length-2;i++){
         int j=i+1,k=nums.length-1;
         while(j<k){
             int s=nums[i]+nums[j]+nums[k];
             if(Math.abs(s-target)<Math.abs(cs-target)){
                 cs=s;
             }
             if(s==target)
             return target;
             else if(s>target)
             k--;
             else
             j++;
         }
     }
     
    return cs;
     
  }

    public static int triangleNumber(int[] nums) { 
     int c=0;
     int n=nums.length;
     Arrays.sort(nums);
     for(int k=n-1;k>1;k--){
         int i=0,j=k-1;
         while(i<j){
             if(nums[i]+nums[j]>nums[k]){
                 c+=j-i;
                 j--;
             }else{
                 i++;
            }
         }
     }
     return c;
  }

  class Solution {
    public int minAddToMakeValid(String s) {
        Stack<Character> st=new Stack<>();
        int c=0;
        for(char ch:s.toCharArray()){
            if(ch=='(')
            st.push(ch);
            else{
                if(st.isEmpty())
                c++;
                else
                st.pop();
            }
        }
        return c+st.size();
    }
}
